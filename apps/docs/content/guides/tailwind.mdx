# Tailwind CSS

`@omnireact/dialog` doesn't render anything itself, so styling the `alert`/`confirm`/
`prompt` templates with Tailwind utility classes is just... writing a normal React
component with `className` props.

## Templates

```tsx
import { DialogsProvider } from '@omnireact/dialog';
import type { ConfirmDialogPayload, DialogProps, DialogTemplates } from '@omnireact/dialog';

function ConfirmDialog({ payload, open, onClose }: DialogProps<ConfirmDialogPayload, boolean>) {
  if (!open) return null;
  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 p-4">
      <div className="w-full max-w-sm rounded-lg border border-neutral-200 bg-white p-5 shadow-xl dark:border-neutral-700 dark:bg-neutral-900">
        {payload.title ? <h3 className="font-semibold">{payload.title}</h3> : null}
        <p className="mt-2 text-sm text-neutral-600 dark:text-neutral-400">{payload.msg}</p>
        <div className="mt-5 flex justify-end gap-2">
          <button
            className="rounded-md px-3 py-1.5 text-sm hover:bg-neutral-100 dark:hover:bg-neutral-800"
            onClick={() => onClose(false)}
          >
            {payload.cancelText ?? 'Cancel'}
          </button>
          <button
            className="rounded-md bg-neutral-900 px-3 py-1.5 text-sm text-white dark:bg-white dark:text-neutral-900"
            onClick={() => onClose(true)}
          >
            {payload.okText ?? 'OK'}
          </button>
        </div>
      </div>
    </div>
  );
}
```

`alert` and `prompt` templates follow the same shape — see the
[API reference](/dialog/api-reference) for their exact payload types.

## Dark mode

This docs site (and this demo) toggle a `dark` class on `<html>` via the theme switcher in
the navbar. If your app does the same, add this once to your global CSS so Tailwind's
`dark:` variant follows that class instead of the OS `prefers-color-scheme` media query:

```css
@import 'tailwindcss';
@custom-variant dark (&:where(.dark, .dark *));
```
